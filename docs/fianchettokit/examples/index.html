    Usage Examples¶

    Copy-paste recipes combining FianchettoKit with ChessCore. They assume a JSONBlobStore named blobStore (see Getting Started).

    A full tactics-attempt flow¶

    Record one attempt against all three tactics stores and the mastery counter, with their shared key:

    import ChessCore
import FianchettoKit

func recordTactic(
    position: Position,
    correctSAN: String,
    solvedCorrectly: Bool,
    elapsed: Double,
    difficulty: PuzzleDifficulty,
    in store: JSONBlobStore
) -> Double {
    let key = BlunderMasteryStore.key(fen: position.fen, correctSAN: correctSAN)
    let outcome: TacticsAttempt.Outcome = solvedCorrectly ? .correct : .wrong

    // 1. Elo.
    var ratings = TacticsRatingStore.load(from: store)
    let delta = ratings.recordAttempt(
        puzzleKey: key, difficulty: difficulty, outcome: outcome, in: store
    )

    // 2. Performance history.
    var perf = TacticsPerformanceStore.load(from: store)
    perf.record(
        TacticsAttempt(
            puzzleKey: key, outcome: outcome, elapsedSeconds: elapsed,
            attemptOrdinal: 1, extractionMode: "punish",
            difficulty: difficulty.rawValue
        ),
        in: store
    )

    // 3. Spaced-repetition mastery.
    BlunderMasteryStore.record(
        fen: position.fen, correctSAN: correctSAN,
        correct: solvedCorrectly, in: store
    )

    return delta   // for the "+12" / "−9" flash
}

    Show the due queue size for the Train hub¶

    func updateTrainHub(masteryKeys: [(fen: String, san: String)], in store: JSONBlobStore) {
    let due = masteryKeys.filter {
        !BlunderMasteryStore.isMastered(fen: $0.fen, correctSAN: $0.san, in: store)
    }.count
    let mastered = masteryKeys.count - due

    TrainProgressCache.record(
        .init(total: masteryKeys.count, due: due, mastered: mastered),
        for: .tactics,
        into: store
    )
}

    Probe a tablebase and pick the move that keeps the win¶

    func tablebaseBestKeepingWin(for position: Position) async throws -> String? {
    guard TablebaseService.pieceCount(for: position) <= 7 else { return nil }
    let result = try await TablebaseService.shared.probe(fen: position.stockfishSafeFEN)
    return result.moves.first { $0.userVerdict == .keepsWin }?.uci
}

    Compare your move against the opening explorer¶

    func explorerPopularity(of san: String, in position: Position) async throws -> Double? {
    let result = try await LichessExplorer.shared.explore(fen: position.fen)
    guard let move = result.moves.first(where: { $0.san == san }),
          result.totalGames > 0 else { return nil }
    return Double(move.totalGames) / Double(result.totalGames)
}

    Import a player's games and prepare them for storage¶

    func importChessCom(username: String) async throws -> [PreparedGameData] {
    let service = ChessAPIService()
    let result = try await service.fetchChessComGames(username: username) { done, total in
        print("archive \(done)/\(total)")
    }
    return result.games.enumerated().map { i, game in
        PreparedGameData(from: game, orderIndex: i)
    }
}

    Drive a board desync gate¶

    BoardSyncGate is a board-agnostic kernel that now lives in BoardKit (renamed from SquareOffSyncGate 2026-07-03) and is re-exported by FianchettoKit — see Stores & the storage seam.

    var gate = BoardSyncGate()

func boardSnapshot(outOfSync: Bool, clockRunning: Bool, toMove: PieceColor) {
    switch gate.update(isOutOfSync: outOfSync,
                       clockIsRunning: clockRunning,
                       currentActiveColor: toMove) {
    case .pauseClock:                pauseTheClock()
    case .resumeClock(let color):    resumeTheClock(for: color)
    case .none:                      break
    }
    if gate.shouldFireHaptic {
        playHaptic()
        gate.acknowledgeHaptic()
    }
}

    Persist a small store over the seam¶

    import FianchettoKit

struct PuzzleProgress: BlobBackedStore {
    static let blobKey = "puzzleProgress"
    var solved: Set<String> = []
    var streak: [Bool] = []
    init() {}
}

let blobStore = MemoryBlobStore()         // your JSONBlobStore implementation
var progress = PuzzleProgress.load(from: blobStore)
progress.solved.insert("puzzle-42")
progress.streak.append(true)
progress.streak.capLast(50)               // keep the most recent 50
progress.save(to: blobStore)

print(StreakMath.current(progress.streak))

import FianchettoKit    // EndgameArchetype

let archetype = EndgameArchetype.allCases.randomElement()!
let fen = archetype.randomFEN()
print(archetype.displayCategory, fen)

if let position = Position(fen: fen) {
    let legal = MoveGenerator.legalMoves(for: position)
    print("\(legal.count) legal moves for \(position.activeColor)")
}
